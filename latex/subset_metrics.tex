\begin{align*}

% --- Precision Fractions ---
\textbf{Global Precision ($P_{\text{global}}$) vs. Subset Precision Estimate ($\widehat{P}_{\text{eval}}$):}

\end{equation}

% --- Lower Bound Proof ---
\textbf{Mathematical Proof of the Precision Lower Bound Constraint:}

\text{Let } \mathcal{F}_{\text{cross}} \text{ be the set of false positive pairs occurring across mismatched artist names:}
\begin{equation}
	\mathcal{F}_{\text{cross}} = \{ (i,j) \in \mathcal{F} \mid \text{Artist}_A \neq \text{Artist}_B \}
\end{equation}

\text{Because your evaluation filter strictly forces identical artist strings, it completely eliminates this entire error pool:}
\begin{equation}
	\mathcal{F}_{\text{cross}} \cap \mathcal{S}_{\text{artist\_clean}} = \emptyset \implies |\mathcal{F} \cap \mathcal{S}_{\text{artist\_clean}}| \ll |\mathcal{F}|
\end{equation}

\text{Now, applying your Concept 1 condition (uniform artificial noise implies that the ratio of model success} \\
\text{on true matches scales linearly with the sample size), we establish the proportional true positive density:}
\begin{equation}
	\frac{|\mathcal{A} \cap \mathcal{M} \cap \mathcal{S}_{\text{artist\_clean}}|}{|\mathcal{A} \cap \mathcal{M}|} \approx \gamma \quad \text{where } \gamma \text{ is the uniform cleanliness retention rate.}
\end{equation}

\text{However, the false positive scaling is heavily broken due to the cross-artist structural exclusion:}
\begin{equation}
	\frac{|\mathcal{F} \cap \mathcal{S}_{\text{artist\_clean}}|}{|\mathcal{F}|} \lll \gamma
\end{equation}

\text{Substituting these asymmetric scaling shifts back into the frequentist denominator reveals the bounding effect:}
\begin{equation}
	\widehat{P}_{\text{eval}} = \frac{\gamma |\mathcal{A} \cap \mathcal{M}|}{\gamma |\mathcal{A} \cap \mathcal{M}| + \epsilon} \quad \text{where } \epsilon = |\mathcal{F} \cap \mathcal{S}_{\text{artist\_clean}}| \to 0
\end{equation}

\text{Because the denominator's error term is suppressed far more aggressively than the numerator's true positive term:}
\begin{equation}
	\widehat{P}_{\text{eval}} \ge P_{\text{global}}
\end{equation}

\textbf{Conclusion:} \text{Your evaluation framework guarantees that subset precision is an optimistic upper bound} \\
\text{for the block, which conversely proves that the true global precision is strictly lower-bounded by your testing reality.}
